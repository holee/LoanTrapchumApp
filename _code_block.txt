@code {
    int selectedIndex = 0;

    RadzenDropDown<string> radzenDropDown = default!;


    private string tempPath = "/assets/logo/tch_Logo.png";

    private string _errorMessage = string.Empty;

    private bool _isError = false;

    private int _progress = 0;

    private bool _isUploading = false;

    private string? _tempPathId = "";

    void OnFileUploadComplete(UploadCompleteEventArgs args)
    {
        if (string.IsNullOrEmpty(args.RawResponse))
        {
            model.ImagePath = tempPath;

            return;
        }

        try
        {
            var options = new JsonSerializerOptions { PropertyNameCaseInsensitive = true };

            var fileModels = JsonSerializer.Deserialize<DocumentDto>(args.RawResponse!, options);

            if (fileModels is not null)
            {
                model.ImagePath = fileModels.FileUrl;

                model.ImagePathId = fileModels.Id;

                _toastService.ShowSuccess("Photo uploaded successfully!", "Upload Complete");
            }
            else
            {
                _isError = true;

                model.ImagePath = tempPath;

                _errorMessage = "Failed to parse upload response.";
            }

            StateHasChanged();
        }
        catch (Exception ex)
        {
            _errorMessage = $"Error parsing upload response: {ex.Message}";

            logger.LogError(ex, "Error parsing upload response: {Message}", ex.Message);
        }
    }

    void OnUploadError(UploadErrorEventArgs args)
    {
        _isError = true;

        _errorMessage = $"Upload failed: {args.Message}";

        _toastService.ShowError(_errorMessage, "Upload Error");

        logger.LogError("Upload failed: {Message}", args.Message);
    }

    void UploadProgress(UploadProgressArgs args)
    {
        _isUploading = true;

        _progress = args.Progress;

        logger.LogInformation("Upload progress: {Progress}%", args.Progress);
    }
}
